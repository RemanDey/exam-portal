One-Sample Hypothesis Test (Z vs. T)
3 functions 📄 api_response.csv Dynamic Z/T selection
Context

A cloud provider claims that the average API response time on their servers is at most 200 ms. A system reliability engineer suspects the true mean is higher and collects a sample of response times. Depending on the sample size, the correct test statistic — either a Z-statistic (n ≥ 30, population std known) or a T-statistic (n < 30, population std unknown) — must be selected dynamically.

File: 📥 api_response.csv Key column: Response Time ms (Float)
A q1A(df)
Extract the Response Time ms column. Using definitional formulas, compute the sample mean (x̄), sample standard deviation (s), and sample size (n). Return these three values in that order.

returns →tuple(float, float, int)
B q1B(x_bar, s, n, mu0, pop_std)
If n ≥ 30 and pop_std > 0, compute the Z-statistic using the known population std. Otherwise compute the T-statistic using the sample std. Compute degrees of freedom (df = n − 1; return 0 for the Z-case).

sfloat — sample standard deviation
nint — sample size

returns →tuple(float, int)

Using a series approximation of the normal CDF (for Z) or the T-distribution CDF via regularised incomplete beta (for T), compute the one-tailed upper-tail p-value. Return True if H₀ is rejected (p-value < alpha), False otherwise.

import numpy as np
import pandas as pd
import math

def q1A(df):
    x = df['Response Time ms'].values
    n = int(len(x))
    x_bar = float(np.sum(x) / n)
    s = float(np.sqrt(np.sum((x - x_bar) ** 2) / (n - 1)))
    return (x_bar, s, n)


def q1B(x_bar, s, n, mu0, pop_std):
    if n >= 30 and pop_std > 0:
        stat = (x_bar - mu0) / (pop_std / np.sqrt(n))
        df = 0
    else:
        stat = (x_bar - mu0) / (s / np.sqrt(n))
        df = n - 1
    return (float(stat), int(df))


def q1C(test_stat, df, alpha):
    if df == 0:                        # Z-test: upper-tail
        p_value = 1.0 - _norm_cdf(test_stat)
    else:                              # T-test: upper-tail
        p_value = _t_sf(test_stat, df)
    decision = bool(p_value < alpha)
    return (float(p_value), decision)


if __name__ == '__main__':
    df = pd.read_csv('api_response.csv')
    x_bar, s, n = q1A(df)
    stat, dof    = q1B(x_bar, s, n, 200.0, 0.0)   # pass actual pop_std if known
    p_val, rej   = q1C(stat, dof, 0.05)
    print(f"x̄={x_bar}, s={s}, n={n}")
    print(f"stat={stat}, df={dof}")
    print(f"p={p_val}, reject={rej}")

# (Assumes _betainc, _betacf, _norm_cdf, _t_sf helpers are defined — see Q1 solution)

def q2A(df):
    a = df['Alpha Latency ms'].values
    b = df['Beta Latency ms'].values
    na, nb  = int(len(a)), int(len(b))
    ma, mb  = float(np.sum(a) / na), float(np.sum(b) / nb)
    va      = float(np.sum((a - ma) ** 2) / (na - 1))
    vb      = float(np.sum((b - mb) ** 2) / (nb - 1))
    return (ma, va, na, mb, vb, nb)


def q2B(mean_alpha, var_alpha, n_alpha, mean_beta, var_beta, n_beta):
    se = np.sqrt(var_alpha / n_alpha + var_beta / n_beta)
    t_stat = float((mean_alpha - mean_beta) / se)

    # Welch-Satterthwaite formula
    term_a = var_alpha / n_alpha
    term_b = var_beta  / n_beta
    num    = (term_a + term_b) ** 2
    denom  = (term_a ** 2 / (n_alpha - 1)) + (term_b ** 2 / (n_beta - 1))
    df     = float(num / denom)
    return (t_stat, df)


def q2C(t_stat, df, alpha):
    # Two-tailed: P(|T| > |t|) = 2 * P(T > |t|)
    p_value  = 2.0 * _t_sf(abs(t_stat), df)
    decision = bool(p_value < alpha)
    return (float(p_value), decision)


if __name__ == '__main__':
    df = pd.read_csv('cluster_latency.csv')
    ma, va, na, mb, vb, nb = q2A(df)
    t_stat, dof             = q2B(ma, va, na, mb, vb, nb)
    p_val, rej              = q2C(t_stat, dof, 0.05)
    print(f"Welch t={t_stat:.10f}, df={dof:.10f}")
    print(f"p={p_val:.10f}, reject={rej}")

Chi-Square Goodness-of-Fit Test
3 functions 📄 network_packets.csv χ² critical value comparison
Context

A network security team monitors traffic at a corporate gateway. According to the vendor specification, packet types should be distributed as: HTTP — 40%, HTTPS — 35%, DNS — 15%, Other — 10%. The team suspects the live distribution has shifted. Compute a Chi-Square Goodness-of-Fit statistic entirely from scratch.

returns →tuple(float, int)
C q3C(chi2_stat, df, chi2_critical)
Compare the test statistic against the critical value (e.g. 7.815 for α=0.05, df=3). Return True if H₀ is rejected, and the absolute difference between the statistic and critical value.
returns →tuple(bool, float)
◈ View Solution
Solution — Q3
def q3A(df):
    observed = {}
    for ptype in ["HTTP", "HTTPS", "DNS", "Other"]:
        observed[ptype] = int(np.sum(df['Packet Type'] == ptype))
    return observed


def q3B(observed, expected_props):
    n      = sum(observed.values())          # total sample size
    chi2   = 0.0
    for key in observed:
        O     = observed[key]
        E     = n * expected_props[key]
        chi2 += (O - E) ** 2 / E
    df = len(observed) - 1                   # 4 categories → df = 3
    return (float(chi2), int(df))


def q3C(chi2_stat, df, chi2_critical):
    rejected = bool(chi2_stat > chi2_critical)
    diff     = float(abs(chi2_stat - chi2_critical))
    return (rejected, diff)


if __name__ == '__main__':
    df     = pd.read_csv('network_packets.csv')
    obs    = q3A(df)
    props  = {"HTTP": 0.40, "HTTPS": 0.35, "DNS": 0.15, "Other": 0.10}
    chi2, dof      = q3B(obs, props)
    rej, diff      = q3C(chi2, dof, 7.815)   # χ²_crit for α=0.05, df=3
    print(f"χ²={chi2:.10f}, df={dof}")
    print(f"reject={rej}, |χ²-crit|={diff:.10f}")

Bootstrap Permutation Test

Context

A pharmaceutical company is testing whether a new formulation reduces average recovery time compared to standard treatment. Since the population distribution is unknown, a Bootstrap Permutation Test is used to assess whether the observed difference in means is statistically significant.

Set random seed. Pool all observations. For each permutation, shuffle with np.random.permutation, split into two groups of original sizes, record the difference in means. Return the permutation distribution as a numpy array.
returns →numpy.ndarray
C q4C(observed_diff, perm_distribution, alpha)
Compute the two-tailed bootstrap p-value as the proportion of permutation differences whose absolute value ≥ |observed_diff|. Return p-value and decision.
returns →tuple(float, bool)
◈ View Solution
Solution — Q4
def q4A(df):
    std_grp = df[df['Treatment'] == 'Standard']['Recovery Days'].values
    new_grp = df[df['Treatment'] == 'New']['Recovery Days'].values
    obs_diff = float(np.sum(new_grp) / len(new_grp)
                   - np.sum(std_grp) / len(std_grp))
    return (obs_diff, std_grp, new_grp)


def q4B(group_standard, group_new, num_permutations, random_seed):
    np.random.seed(random_seed)
    combined  = np.concatenate([group_standard, group_new])
    n_std     = len(group_standard)
    perm_diff = np.empty(num_permutations)
    for i in range(num_permutations):
        shuffled       = np.random.permutation(combined)
        perm_std       = shuffled[:n_std]
        perm_new       = shuffled[n_std:]
        perm_diff[i]   = (np.sum(perm_new) / len(perm_new)
                        - np.sum(perm_std) / len(perm_std))
    return perm_diff


def q4C(observed_diff, perm_distribution, alpha):
    p_value  = float(
        np.sum(np.abs(perm_distribution) >= abs(observed_diff))
        / len(perm_distribution)
    )
    decision = bool(p_value < alpha)
    return (p_value, decision)


if __name__ == '__main__':
    df = pd.read_csv('recovery_times.csv')
    obs_diff, std_grp, new_grp = q4A(df)
    perm_dist                  = q4B(std_grp, new_grp, 10000, 42)
    p_val, rej                 = q4C(obs_diff, perm_dist, 0.05)
    print(f"Observed diff={obs_diff:.10f}")
    print(f"p={p_val:.10f}, reject={rej}")

Type I Error, Type II Error & Statistical Power
3 functions Simulation-based · No CSV Empirical vs Theoretical Power
Context

A factory quality-control system flags a batch as defective if the sample mean weight falls below a threshold μ₀. Empirically estimate the Type I error rate (false alarm), Type II error rate (missed detection), and statistical power through simulation, and compare empirical power against the theoretical formula.

A q5A(num_simulations, random_seed, mu0, sigma, n, alpha, z_critical)
Under H₀ (true mean = mu0), draw num_simulations samples of size n from a normal distribution. Compute Z-statistic for each sample. The empirical Type I error rate is the proportion where H₀ is incorrectly rejected (Z < z_critical for lower-tail test).
returns →float
B q5B(num_simulations, random_seed, mu0, sigma, n, alpha, z_critical, mu1)
Under H₁ (true mean = mu1), draw num_simulations samples of size n. The empirical Type II error rate (β) is the proportion where H₀ is NOT rejected even though it is false.
returns →float
C q5C(num_simulations, random_seed, mu0, sigma, n, alpha, z_critical, mu1)
Derive empirical power = 1 − β. Compute theoretical power:
Power = Φ(z_critical + (mu0 − mu1) / (sigma / √n))
Return empirical power, theoretical power, and absolute difference between them.
returns →tuple(float, float, float)
◈ View Solution
Solution — Q5
# (Assumes _norm_cdf helper is defined — see Q1 solution)

def q5A(num_simulations, random_seed, mu0, sigma, n, alpha, z_critical):
    np.random.seed(random_seed)
    # Draw all samples at once for efficiency
    samples      = np.random.normal(mu0, sigma, (num_simulations, n))
    sample_means = np.sum(samples, axis=1) / n
    z_stats      = (sample_means - mu0) / (sigma / np.sqrt(n))
    # Lower-tail test: reject H0 when Z < z_critical (z_critical is negative, e.g. -1.645)
    type1_rate   = float(np.sum(z_stats < z_critical) / num_simulations)
    return type1_rate


def q5B(num_simulations, random_seed, mu0, sigma, n, alpha, z_critical, mu1):
    np.random.seed(random_seed)
    samples      = np.random.normal(mu1, sigma, (num_simulations, n))
    sample_means = np.sum(samples, axis=1) / n
    z_stats      = (sample_means - mu0) / (sigma / np.sqrt(n))
    # Type II: fail to reject H0 (Z >= z_critical) even though H0 is false
    beta         = float(np.sum(z_stats >= z_critical) / num_simulations)
    return beta


def q5C(num_simulations, random_seed, mu0, sigma, n, alpha, z_critical, mu1):
    beta         = q5B(num_simulations, random_seed, mu0, sigma, n,
                       alpha, z_critical, mu1)
    emp_power    = float(1.0 - beta)
    # Theoretical power: Phi(z_critical + (mu0 - mu1) / SE)
    se           = sigma / np.sqrt(n)
    z_arg        = z_critical + (mu0 - mu1) / se
    theo_power   = float(_norm_cdf(z_arg))
    diff         = float(abs(emp_power - theo_power))
    return (emp_power, theo_power, diff)


if __name__ == '__main__':
    params = dict(num_simulations=50000, random_seed=0,
                  mu0=500.0, sigma=10.0, n=30, alpha=0.05,
                  z_critical=-1.645)
    t1  = q5A(**params)
    t2  = q5B(**params, mu1=495.0)
    ep, tp, d = q5C(**params, mu1=495.0)
    print(f"Type I  = {t1:.10f}")
    print(f"Type II = {t2:.10f}")
    print(f"Emp Power={ep:.10f}, Theo Power={tp:.10f}, |diff|={d:.10f}")

One-Way ANOVA (Analysis of Variance)

An electronics manufacturer tests three battery suppliers — Supplier X, Y, and Z — to determine whether mean battery life differs significantly across suppliers. Implement a complete one-way ANOVA from scratch: decompose total variance into between-group and within-group components, compute the F-statistic, and reach a test decision.

Calculate SSB (Sum of Squares Between) and SSW (Sum of Squares Within). Derive df_between = k−1, df_within = N−k. Compute MSB, MSW, and F-statistic = MSB/MSW. Return SSB, SSW, df_between, df_within, F.
returns →tuple(float, float, int, int, float)
C q6C(f_stat, df_pair, f_critical)
Compare F-statistic against the critical F-value (e.g. 3.056 for α=0.05, df=(2,147)). Return True if H₀ is rejected (at least one group mean differs), and the absolute difference F_stat − F_critical.

Solution — Q6
def q6A(df):
    x_vals     = df[df['Supplier'] == 'X']['Battery Life Hours'].values
    y_vals     = df[df['Supplier'] == 'Y']['Battery Life Hours'].values
    z_vals     = df[df['Supplier'] == 'Z']['Battery Life Hours'].values
    mean_X     = float(np.sum(x_vals) / len(x_vals))
    mean_Y     = float(np.sum(y_vals) / len(y_vals))
    mean_Z     = float(np.sum(z_vals) / len(z_vals))
    all_vals   = np.concatenate([x_vals, y_vals, z_vals])
    grand_mean = float(np.sum(all_vals) / len(all_vals))
    return (mean_X, mean_Y, mean_Z, grand_mean)


def q6B(df):
    x_vals   = df[df['Supplier'] == 'X']['Battery Life Hours'].values
    y_vals   = df[df['Supplier'] == 'Y']['Battery Life Hours'].values
    z_vals   = df[df['Supplier'] == 'Z']['Battery Life Hours'].values
    groups   = [x_vals, y_vals, z_vals]
    all_vals = np.concatenate(groups)
    N        = len(all_vals)
    k        = 3

    grand_mean = float(np.sum(all_vals) / N)

    # SSB: Σ n_i * (x̄_i − x̄)²
    SSB = float(sum(
        len(g) * (float(np.sum(g) / len(g)) - grand_mean) ** 2
        for g in groups
    ))
    # SSW: Σ Σ (x_ij − x̄_i)²
    SSW = float(sum(
        float(np.sum((g - np.sum(g) / len(g)) ** 2))
        for g in groups
    ))

    df_between = int(k - 1)
    df_within  = int(N - k)
    MSB        = SSB / df_between
    MSW        = SSW / df_within
    F          = float(MSB / MSW)
    return (SSB, SSW, df_between, df_within, F)


def q6C(f_stat, df_pair, f_critical):
    rejected = bool(f_stat > f_critical)
    diff     = float(abs(f_stat - f_critical))
    return (rejected, diff)


if __name__ == '__main__':
    df = pd.read_csv('battery_life.csv')
    mX, mY, mZ, gm     = q6A(df)
    SSB, SSW, dfb, dfw, F = q6B(df)
    rej, diff           = q6C(F, (dfb, dfw), 3.056)
    print(f"Means: X={mX:.10f}, Y={mY:.10f}, Z={mZ:.10f}, Grand={gm:.10f}")
    print(f"SSB={SSB:.10f}, SSW={SSW:.10f}")
    print(f"df_b={dfb}, df_w={dfw}, F={F:.10f}")
    print(f"reject={rej}, |F-F_crit|={diff:.10f}")
